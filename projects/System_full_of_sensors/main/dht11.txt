#include <stdio.h>

#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#include "driver/gpio.h"
#include "esp_err.h"
#include <dht.h>

#define DHT_TYPE DHT_TYPE_DHT11
#define DHT_PIN GPIO_NUM_4

void app_main(void)
{
    float humidity = 0.0;
    float temperature = 0.0;
    while (1)
    {
        esp_err_t result = dht_read_float_data(DHT_TYPE, DHT_PIN, &humidity, &temperature);
        if (result == ESP_OK)
        {
            printf("Humidity: %.1f%%  Temperature: %.1fC\n", humidity, temperature);
        }
        else
        {
            printf("Could not read data from sensor: %s\n", esp_err_to_name(result));
        }
        vTaskDelay(2000 / portTICK_PERIOD_MS);
    }
}
